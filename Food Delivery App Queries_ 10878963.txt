-- Q1. Display all registered customers

SELECT CUSTOMER_ID, FULLNAME, EMAIL, MOBILE, IS_ACTIVE, CREATED_AT
FROM CUSTOMER;
 
-- Q2. Display all active restaurants

SELECT RESTAURANT_ID, RESTAURANT_NAME, AREA, CITY, PHONE, EMAIL
FROM RESTAURANT
WHERE IS_ACTIVE = 1;
 
-- Q3. Find all vegetarian food items

SELECT ITEM_ID, ITEM_NAME, ITEM_DESCRIPTION, PRICE
FROM MENU_ITEM
WHERE IS_VEG = 1;
 
-- Q4. Display food items belonging to a particular restaurant (SPICE GARDEN)

SELECT R.RESTAURANT_NAME, M.ITEM_NAME, M.ITEM_DESCRIPTION, M.PRICE, M.IS_AVAILABLE
FROM MENU_ITEM M
INNER JOIN RESTAURANT R ON R.RESTAURANT_ID = M.RESTAURANT_ID
WHERE R.RESTAURANT_NAME = 'SPICE GARDEN';
 
-- Q5. Find food items whose price is greater than Rs.200

SELECT ITEM_ID, ITEM_NAME, PRICE
FROM MENU_ITEM
WHERE PRICE > 200;
 
-- Q6. Display all orders placed by a particular customer (CUSTOMER_ID = 1, MAHESH)

SELECT O.ORDER_ID, O.ORDER_DATE, O.TOTAL_AMOUNT, O.ORDER_STATUS
FROM ORDERS O
WHERE O.CUSTOMER_ID = 1;
 
-- Q7. Find all orders placed on a particular date (18-Oct-2026)

SELECT ORDER_ID, CUSTOMER_ID, RESTAURANT_ID, ORDER_DATE, TOTAL_AMOUNT, ORDER_STATUS
FROM ORDERS
WHERE ORDER_DATE >= '2026-10-18' AND ORDER_DATE < '2026-10-19';
 
-- Q8. Display all available delivery partners

SELECT PARTNER_ID, PARTNER_NAME, MOBILE, VEHICLE_TYPE, VEHICLE_NUMBER
FROM DELIVERY_PARTNER
WHERE AVAILABILITY_STATUS = 'AVAILABLE';

-- INTERMEDIATE LEVEL
-- #####################################################################
 
-- Q9. Display customer details along with their delivery addresses

SELECT C.CUSTOMER_ID, C.FULLNAME, C.EMAIL, C.MOBILE,
       A.ADDRESS_LINE, A.AREA, A.CITY, A.PINCODE
FROM CUSTOMER C
INNER JOIN CUSTOMER_ADDRESS A ON A.CUSTOMER_ID = C.CUSTOMER_ID;
 
-- Q10. Display restaurant names along with their food menu

SELECT R.RESTAURANT_NAME, M.ITEM_NAME, F.NAME_CATEGORY, M.PRICE
FROM RESTAURANT R
LEFT JOIN MENU_ITEM M     ON M.RESTAURANT_ID = R.RESTAURANT_ID
LEFT JOIN FOOD_CATEGORY F ON F.CATEGORY_ID   = M.CATEGORY_ID
ORDER BY R.RESTAURANT_NAME;
 
-- Q11. Display complete order details including customer and restaurant information

SELECT O.ORDER_ID, O.ORDER_DATE,
       C.FULLNAME AS CUSTOMER_NAME, C.MOBILE,
       A.ADDRESS_LINE AS DELIVERY_ADDRESS,
       R.RESTAURANT_NAME,
       M.ITEM_NAME, OI.QUANTITY, OI.UNIT_PRICE,
       O.SUBTOTAL, O.DELIVERY_CHARGE, O.DISCOUNT_AMOUNT, O.TOTAL_AMOUNT, O.ORDER_STATUS
FROM ORDERS O
INNER JOIN CUSTOMER C          ON C.CUSTOMER_ID   = O.CUSTOMER_ID
INNER JOIN CUSTOMER_ADDRESS A  ON A.ADDRESS_ID    = O.ADDRESS_ID
INNER JOIN RESTAURANT R        ON R.RESTAURANT_ID = O.RESTAURANT_ID
INNER JOIN ORDER_ITEM OI       ON OI.ORDER_ID     = O.ORDER_ID
INNER JOIN MENU_ITEM M         ON M.ITEM_ID       = OI.ITEM_ID
ORDER BY O.ORDER_ID;
 
-- Q12. Find the total number of orders placed by each customer

SELECT C.CUSTOMER_ID, C.FULLNAME, COUNT(O.ORDER_ID) AS TOTAL_ORDERS
FROM CUSTOMER C
LEFT JOIN ORDERS O ON O.CUSTOMER_ID = C.CUSTOMER_ID
GROUP BY C.CUSTOMER_ID, C.FULLNAME
ORDER BY TOTAL_ORDERS DESC;
 
-- Q13. Find the total number of food items available in each restaurant

SELECT R.RESTAURANT_ID, R.RESTAURANT_NAME, COUNT(M.ITEM_ID) AS AVAILABLE_ITEMS
FROM RESTAURANT R
LEFT JOIN MENU_ITEM M ON M.RESTAURANT_ID = R.RESTAURANT_ID
                     AND M.IS_AVAILABLE = 1
GROUP BY R.RESTAURANT_ID, R.RESTAURANT_NAME;
 
-- Q14. Calculate the total sales generated by each restaurant

SELECT R.RESTAURANT_ID, R.RESTAURANT_NAME, ISNULL(SUM(O.TOTAL_AMOUNT), 0) AS TOTAL_SALES
FROM RESTAURANT R
LEFT JOIN ORDERS O ON O.RESTAURANT_ID = R.RESTAURANT_ID
                  AND O.ORDER_STATUS <> 'CANCELLED'
GROUP BY R.RESTAURANT_ID, R.RESTAURANT_NAME
ORDER BY TOTAL_SALES DESC;
 
-- Q15. Find the most frequently ordered food item (by quantity, excluding cancelled orders)

SELECT TOP 1 WITH TIES
       M.ITEM_ID, M.ITEM_NAME, SUM(OI.QUANTITY) AS TOTAL_QUANTITY_ORDERED
FROM ORDER_ITEM OI
INNER JOIN ORDERS O    ON O.ORDER_ID = OI.ORDER_ID AND O.ORDER_STATUS <> 'CANCELLED'
INNER JOIN MENU_ITEM M ON M.ITEM_ID  = OI.ITEM_ID
GROUP BY M.ITEM_ID, M.ITEM_NAME
ORDER BY SUM(OI.QUANTITY) DESC;
 
-- Q16. Find customers who have never placed an order

SELECT C.CUSTOMER_ID, C.FULLNAME, C.EMAIL
FROM CUSTOMER C
LEFT JOIN ORDERS O ON O.CUSTOMER_ID = C.CUSTOMER_ID
WHERE O.ORDER_ID IS NULL;
 
-- Q17. Display delivery partners who have completed more than five deliveries
--      (with the sample data nobody has more than 5, so this returns no rows)

SELECT P.PARTNER_ID, P.PARTNER_NAME, COUNT(D.DELIVERY_ID) AS COMPLETED_DELIVERIES
FROM DELIVERY_PARTNER P
INNER JOIN DELIVERY D ON D.PARTNER_ID = P.PARTNER_ID
                     AND D.ASSIGNMENT_STATUS = 'DELIVERED'
GROUP BY P.PARTNER_ID, P.PARTNER_NAME
HAVING COUNT(D.DELIVERY_ID) > 5;
 
-- Q17 (demo with lower limit, to show the query works: more than 2 deliveries)

SELECT P.PARTNER_ID, P.PARTNER_NAME, COUNT(D.DELIVERY_ID) AS COMPLETED_DELIVERIES

-- Q18. Find the average rating of each restaurant

SELECT R.RESTAURANT_ID, R.RESTAURANT_NAME,
       CAST(AVG(RV.RATING * 1.0) AS DECIMAL(3,2)) AS AVG_RATING,
       COUNT(RV.REVIEW_ID) AS TOTAL_REVIEWS
FROM RESTAURANT R
LEFT JOIN REVIEW RV ON RV.RESTAURANT_ID = R.RESTAURANT_ID
                   AND RV.ITEM_ID IS NULL
GROUP BY R.RESTAURANT_ID, R.RESTAURANT_NAME;

SELECT TOP 1 WITH TIES
       C.CUSTOMER_ID, C.FULLNAME, SUM(O.TOTAL_AMOUNT) AS TOTAL_SPENT

-- Q20. Find the restaurant that generated the highest revenue

SELECT TOP 1 WITH TIES
       R.RESTAURANT_ID, R.RESTAURANT_NAME, SUM(O.TOTAL_AMOUNT) AS TOTAL_REVENUE

INNER JOIN ORDERS O ON O.RESTAURANT_ID = R.RESTAURANT_ID

-- Q21. Find the second most expensive food item

SELECT ITEM_ID, ITEM_NAME, PRICE
FROM MENU_ITEM
WHERE PRICE = (SELECT MAX(PRICE)
               FROM MENU_ITEM
               WHERE PRICE < (SELECT MAX(PRICE) FROM MENU_ITEM));
 
-- Q22. Find restaurants whose average rating is greater than the overall average rating

SELECT R.RESTAURANT_ID, R.RESTAURANT_NAME,
       CAST(AVG(RV.RATING * 1.0) AS DECIMAL(3,2)) AS AVG_RATING
FROM RESTAURANT R
INNER JOIN REVIEW RV ON RV.RESTAURANT_ID = R.RESTAURANT_ID
                    AND RV.ITEM_ID IS NULL
GROUP BY R.RESTAURANT_ID, R.RESTAURANT_NAME
HAVING AVG(RV.RATING * 1.0) > (SELECT AVG(RATING * 1.0)
                               FROM REVIEW
                               WHERE ITEM_ID IS NULL);
 
 
-- Q23. Find customers who have placed more orders than the average orders per customer

SELECT C.CUSTOMER_ID, C.FULLNAME, COUNT(O.ORDER_ID) AS TOTAL_ORDERS
FROM CUSTOMER C
INNER JOIN ORDERS O ON O.CUSTOMER_ID = C.CUSTOMER_ID
GROUP BY C.CUSTOMER_ID, C.FULLNAME
HAVING COUNT(O.ORDER_ID) > (SELECT COUNT(*) * 1.0 / (SELECT COUNT(*) FROM CUSTOMER)
                            FROM ORDERS);
 
-- Q24. Find food items that have never been ordered

SELECT M.ITEM_ID, M.ITEM_NAME, M.PRICE
FROM MENU_ITEM M
WHERE NOT EXISTS (SELECT 1 FROM ORDER_ITEM OI WHERE OI.ITEM_ID = M.ITEM_ID);
 
-- Q25. Find the top three best-selling food items (by quantity, excluding cancelled orders)

SELECT TOP 3
       M.ITEM_ID, M.ITEM_NAME, SUM(OI.QUANTITY) AS TOTAL_QUANTITY_SOLD
FROM ORDER_ITEM OI
INNER JOIN ORDERS O    ON O.ORDER_ID = OI.ORDER_ID AND O.ORDER_STATUS <> 'CANCELLED'
INNER JOIN MENU_ITEM M ON M.ITEM_ID  = OI.ITEM_ID
GROUP BY M.ITEM_ID, M.ITEM_NAME
ORDER BY SUM(OI.QUANTITY) DESC;
 
-- Q26. Identify customers who have used promotional coupons

SELECT C.CUSTOMER_ID, C.FULLNAME, CP.COUPON_CODE, CR.DISCOUNT_APPLIED, CR.REDEEMED_AT
FROM COUPON_REDEMPTION CR
INNER JOIN CUSTOMER C ON C.CUSTOMER_ID = CR.CUSTOMER_ID
INNER JOIN COUPON CP  ON CP.COUPON_ID  = CR.COUPON_ID;
 
-- Q27. Find restaurants that have not received any orders

SELECT R.RESTAURANT_ID, R.RESTAURANT_NAME
FROM RESTAURANT R
WHERE NOT EXISTS (SELECT 1 FROM ORDERS O WHERE O.RESTAURANT_ID = R.RESTAURANT_ID);
 
-- Q28. Generate monthly restaurant revenue report

SELECT YEAR(O.ORDER_DATE)  AS ORDER_YEAR,
       MONTH(O.ORDER_DATE) AS ORDER_MONTH,
       R.RESTAURANT_NAME,
       COUNT(O.ORDER_ID)     AS TOTAL_ORDERS,
       SUM(O.TOTAL_AMOUNT)   AS MONTHLY_REVENUE
FROM ORDERS O
INNER JOIN RESTAURANT R ON R.RESTAURANT_ID = O.RESTAURANT_ID
WHERE O.ORDER_STATUS <> 'CANCELLED'
GROUP BY YEAR(O.ORDER_DATE), MONTH(O.ORDER_DATE), R.RESTAURANT_NAME
ORDER BY ORDER_YEAR, ORDER_MONTH, R.RESTAURANT_NAME;
 
-- Q29. Find delivery partners with the highest number of completed deliveries

SELECT TOP 1 WITH TIES
       P.PARTNER_ID, P.PARTNER_NAME, COUNT(D.DELIVERY_ID) AS COMPLETED_DELIVERIES
FROM DELIVERY_PARTNER P
INNER JOIN DELIVERY D ON D.PARTNER_ID = P.PARTNER_ID
                     AND D.ASSIGNMENT_STATUS = 'DELIVERED'
GROUP BY P.PARTNER_ID, P.PARTNER_NAME
ORDER BY COUNT(D.DELIVERY_ID) DESC;
 
-- Q30. Identify customers who have placed orders but have never submitted a review

SELECT C.CUSTOMER_ID, C.FULLNAME
FROM CUSTOMER C
WHERE EXISTS     (SELECT 1 FROM ORDERS O  WHERE O.CUSTOMER_ID  = C.CUSTOMER_ID)
  AND NOT EXISTS (SELECT 1 FROM REVIEW RV WHERE RV.CUSTOMER_ID = C.CUSTOMER_ID);
