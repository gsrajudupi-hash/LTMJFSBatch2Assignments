-- CREATED TABLE ORDER --


CREATE TABLE ORDERS (
 ORDER_ID INT AUTO_INCREMENT PRIMARY KEY,
 CUSTOMER_ID INT NOT NULL,
 RESTAURANT_ID INT NOT NULL,
 ADDRESS_ID INT NOT NULL,
 COUPON_ID INT NULL,
 ORDER_DATE DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
 SUBTOTAL DECIMAL(10,2) NOT NULL,
 DELIVERY_CHARGE DECIMAL(8,2) NOT NULL DEFAULT 0,
 DISCOUNT_AMOUNT DECIMAL(8,2) NOT NULL DEFAULT 0,
 TOTAL_AMOUNT DECIMAL(10,2) NOT NULL,

 FOREIGN KEY (CUSTOMER_ID)   REFERENCES CUSTOMER(CUSTOMER_ID),
 FOREIGN KEY (RESTAURANT_ID) REFERENCES RESTAURANT(RESTAURANT_ID),
 FOREIGN KEY (ADDRESS_ID)    REFERENCES CUSTOMER_ADDRESS(ADDRESS_ID),
 FOREIGN KEY (COUPON_ID)     REFERENCES COUPON(COUPON_ID)
);

-- CREATED A REVIEW  TABLE --


CREATE TABLE REVIEW (
 REVIEW_ID INT AUTO_INCREMENT PRIMARY KEY,
 CUSTOMER_ID INT NOT NULL,
 ORDER_ID INT NOT NULL,
 RESTAURANT_ID INT NOT NULL,
 ITEM_ID INT NULL,
 RATING INT NOT NULL,
 REVIEW_TEXT VARCHAR(500) NULL,
 CREATED_AT DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP(),
 UNIQUE (ORDER_ID, ITEM_ID),
 FOREIGN KEY (CUSTOMER_ID)   REFERENCES CUSTOMER(CUSTOMER_ID),
 FOREIGN KEY (ORDER_ID)      REFERENCES ORDERS(ORDER_ID),
 FOREIGN KEY (RESTAURANT_ID) REFERENCES RESTAURANT(RESTAURANT_ID),
 FOREIGN KEY (ITEM_ID)       REFERENCES MENU_ITEM(ITEM_ID)
);
